<script>
    // MISMA CONFIGURACIÓN EXACTA QUE EN EL CÓDIGO QUE SÍ FUNCIONA
    const CREDENCIALES = {
        usuario: "srchaconm@gmail.com",
        clave: "ContraseñaSegura1"
    };
    const SUPABASE_URL = "https://swwodfpirqvwliuiswup.supabase.co";
    const SUPABASE_KEY = "sb_publishable_FKCzw_BciB92cN3VNFLevA_rz49uq-q";

    const clienteSupabase = supabase.createClient(SUPABASE_URL, SUPABASE_KEY, {
        auth: { storage: localStorage, autoRefreshToken: true, persistSession: true, detectSessionInUrl: false }
    });

    // MISMA LÓGICA DE INICIO
    function mostrarSistema() {
        document.getElementById('cargando').style.display = 'none';
        document.getElementById('formIngreso').style.display = 'block';
        contarEmpresas();
        cargarNIT();
    }

    async function contarEmpresas() {
        try {
            const { count, error } = await clienteSupabase
                .from('Empresa')
                .select('*', { count: 'exact', head: true });

            if (error) throw error;
            
            document.getElementById('mensajeConteo').textContent = `✅ Existen ${count} empresas registradas`;
            document.getElementById('mensajeConteo').style.display = 'block';
        } catch (err) {
            console.error('Error conteo:', err);
            document.getElementById('mensajeConteo').textContent = `⚠️ No se pudo obtener el conteo: ${err.message}`;
            document.getElementById('mensajeConteo').style.display = 'block';
        }
    }

    async function cargarNIT() {
    try {
        // ✅ Se usa comillas dobles porque el campo tiene espacio: "Nombre Legal"
        const { data, error } = await clienteSupabase
            .from('Empresa')
            .select("NIT, \"Nombre Legal\"")
            .order("NIT", { ascending: true });

        if (error) throw error;

        const selectNit = document.getElementById('nit');
        selectNit.innerHTML = '<option value="">Seleccione una empresa</option>';
        data.forEach(item => {
            let opcion = document.createElement('option');
            opcion.value = item.NIT;
            // ✅ Se accede al campo con espacio usando corchetes
            opcion.textContent = `${item.NIT} - ${item["Nombre Legal"] || 'Sin nombre'}`;
            selectNit.appendChild(opcion);
        });

    } catch (err) {
        console.error('Error al cargar empresas:', err);
        alert(`Error al cargar: ${err.message}`);
    }
}
    async function iniciarSesionAutomatica() {
        try {
            const { data: { session } } = await clienteSupabase.auth.getSession();
            if (session) { mostrarSistema(); return; }
            
            const { error } = await clienteSupabase.auth.signInWithPassword({
                email: CREDENCIALES.usuario, 
                password: CREDENCIALES.clave
            });
            if (error) throw error;
            
            mostrarSistema();
        } catch (err) {
            document.getElementById('cargando').innerHTML = `<p class="error">❌ Error de conexión: ${err.message}</p>`;
            console.error("Fallo autenticación:", err);
        }
    }

    window.addEventListener('load', iniciarSesionAutomatica);

    const form = document.getElementById('formIngreso');
    form.addEventListener('submit', function(e) {
        e.preventDefault();
        const nit = document.getElementById('nit').value;
        let valido = true;

        if (!nit) {
            document.getElementById('errorNit').style.display = 'block';
            valido = false;
        } else {
            document.getElementById('errorNit').style.display = 'none';
        }

        if (valido) {
            window.location.href = `MantenimientoSucursal.html?nit=${encodeURIComponent(nit)}`;
        }
    });
